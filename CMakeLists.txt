cmake_minimum_required(VERSION 3.16)
project(NetScope VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(netscope_core src/parser.cpp src/statistics.cpp)
target_include_directories(netscope_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(netscope_core PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)

find_package(Threads REQUIRED)
find_path(PCAP_INCLUDE_DIR pcap.h)
find_library(PCAP_LIBRARY NAMES pcap)
if(PCAP_INCLUDE_DIR AND PCAP_LIBRARY)
  add_executable(netscope src/main.cpp)
  target_include_directories(netscope PRIVATE ${PCAP_INCLUDE_DIR})
  target_link_libraries(netscope PRIVATE netscope_core Threads::Threads ${PCAP_LIBRARY})
  target_compile_options(netscope PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
else()
  message(WARNING "libpcap development files not found: building parser tests only")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(netscope_tests tests/parser_tests.cpp)
  target_link_libraries(netscope_tests PRIVATE netscope_core Threads::Threads)
  add_test(NAME parser_tests COMMAND netscope_tests)
  add_executable(netscope_pipeline_tests tests/pipeline_tests.cpp)
  target_link_libraries(netscope_pipeline_tests PRIVATE netscope_core Threads::Threads)
  add_test(NAME pipeline_tests COMMAND netscope_pipeline_tests)
  if(TARGET netscope)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      add_test(NAME offline_smoke COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/offline_smoke.py $<TARGET_FILE:netscope>)
      add_test(NAME offline_load COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/offline_load.py $<TARGET_FILE:netscope>)
      add_test(NAME demo_capture COMMAND $<TARGET_FILE:netscope>
        --file ${CMAKE_CURRENT_SOURCE_DIR}/samples/demo.pcap)
      set_tests_properties(demo_capture PROPERTIES
        PASS_REGULAR_EXPRESSION "ARP: 1  TCP: 1  UDP: 1  ICMP: 1")
    endif()
  endif()
endif()
